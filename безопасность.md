1) Убрать вход по паролю. Использовать ssh ключи (ed25519)
2) Запретить вход root пользователя. Создать sudo пользователя
3) вход только по ssh ключу через пользователя

0) открыть ssh сессию через root и не закрывать, пока не убедишься, что все верно настроил
1) поменять пароль root
passwd root
2) создать нового пользователя
adduser username
3) добавить в группу sudo
usermod -aG sudo username


создание публичного ssh ключа на удаленном сервере
su - username
mkdir -p ~/.ssh
chmod 700 ~/.ssh
nano ~/.ssh/authorized_keys

На локальной машине создаем пару ssh ключей
ssh-keygen -t ed25519 -C "vps" -f ~/.ssh/vps_ed25519


Копирование ПУБЛИЧНОГО ключа:
cat ~/.ssh/vps_ed25519.pub

вставляем ключ на удаленном сервере nano ~/.ssh/authorized_keys
chmod 600 ~/.ssh/authorized_keys
sudo nano ~/.ssh/authorized_keys

nano /etc/ssh/sshd_config
запретить вход для root
PermitRootLogin no
PasswordAuthentication no

sshd -t && systemctl restart ssh


Копирование конфигов с vps. Помним, что вход через root = бан (fail2ban)
sudo cp -r /root/awg/ /home/myusername/
sudo chown -R myusername:myusername /home/myusername/awg/

scp -r vps2:/home/myusername/awg/ .

На локальной машине для удобства:
nano ~/.ssh/config
Host vps2
    HostName 256.77.111.67
    User myusername
    IdentityFile ~/.ssh/vps2_ed25519
    IdentitiesOnly yes

вход по ssh: ssh vps2
